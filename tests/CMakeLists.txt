cmake_minimum_required(VERSION 3.13)
project(wp2_host_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
get_filename_component(WP2_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(MINIZ "${WP2_ROOT}/app/sdk/ui/svgrender/miniz-3.0.2")
foreach(test panel controls crown)
  add_executable(test_${test} test_${test}.c)
  add_test(NAME ${test} COMMAND test_${test})
endforeach()
if(EXISTS "${WP2_ROOT}/generated/badapple_data.c")
  add_executable(test_video test_video.c "${WP2_ROOT}/app/src/video/player.c"
    "${WP2_ROOT}/generated/badapple_data.c" "${MINIZ}/miniz_tinfl.c")
  target_include_directories(test_video PRIVATE "${MINIZ}")
  target_compile_definitions(test_video PRIVATE MINIZ_NO_MALLOC MINIZ_NO_STDIO MINIZ_NO_TIME)
  if(EXISTS "${WP2_ROOT}/assets/private/source_192x144_15fps.gray")
    add_test(NAME video COMMAND test_video "${WP2_ROOT}/assets/private/source_192x144_15fps.gray")
  endif()
endif()
add_executable(test_audio test_audio.c "${WP2_ROOT}/app/src/audio/codec.c")
target_include_directories(test_audio PRIVATE "${WP2_ROOT}/third_party/minimp3")
if(EXISTS "${WP2_ROOT}/assets/private/badapple_song_16k.mp3")
  add_test(NAME audio COMMAND test_audio "${WP2_ROOT}/assets/private/badapple_song_16k.mp3")
endif()
